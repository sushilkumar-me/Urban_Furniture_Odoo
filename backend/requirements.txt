
# FastAPI - the web framework

uvicorn==0.30.1

# SQLAlchemy - ORM to talk to PostgreSQL using Python classes
sqlalchemy==2.0.31

# Python-dotenv - reads the .env file
python-dotenv==1.0.1

# Passlib - password hashing library.
# We use it to hash passwords before saving to the database.
# It wraps bcrypt so we never store plain-text passwords.
# [bcrypt] means: install passlib WITH bcrypt support included.
passlib[bcrypt]==1.7.4

# Bcrypt - the actual hashing algorithm used by passlib.
# bcrypt is intentionally SLOW — it makes brute-force attacks
# computationally expensive. That is a feature, not a bug.
bcrypt==4.0.1

# Python-jose - creates and verifies JWT tokens.
# JWT = JSON Web Token. After login, we give the user a token.
# On every future request, they send the token to prove identity.
# We don't need to hit the database to verify them each time.
# [cryptography] means: install with cryptography backend support.
python-jose[cryptography]==3.3.0

faker>=30.0.0
